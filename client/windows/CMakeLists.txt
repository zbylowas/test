cmake_minimum_required(VERSION 3.21)
project(vipnet_diagnostics VERSION 0.5.2 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows SDK and MSVC required")
endif()
enable_language(RC)
configure_file(version.rc.in version.rc @ONLY)
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(zlib URL https://zlib.net/zlib-1.4.tar.gz https://zlib.net/fossils/zlib-1.4.tar.gz
  URL_HASH SHA256=02dd159783677a749c04adbdd02b55b8a2bcc7e054f51b76b549199a916ad584)
FetchContent_MakeAvailable(zlib)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(nlohmann_json)
add_library(diagnostics_core STATIC main.cpp icmp.cpp tcp.cpp queue.cpp sender.cpp throughput.cpp inventory.cpp inventory_support.cpp wifi.cpp)
add_executable(vipnet-mtr entry.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_link_libraries(vipnet-mtr PRIVATE diagnostics_core)
add_executable(VIP-NET-Diagnostyka WIN32 desktop.cpp speed_chart.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_link_libraries(VIP-NET-Diagnostyka PRIVATE diagnostics_core nlohmann_json::nlohmann_json)
target_compile_features(VIP-NET-Diagnostyka PRIVATE cxx_std_20)
target_compile_definitions(VIP-NET-Diagnostyka PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(VIP-NET-Diagnostyka PRIVATE /W4 /utf-8 /permissive-)
add_executable(vipnet-tcp-check tcp_check.cpp icmp.cpp tcp.cpp)
foreach(binary IN ITEMS diagnostics_core vipnet-tcp-check)
  target_compile_features(${binary} PRIVATE cxx_std_20)
  target_compile_definitions(${binary} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(${binary} PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET ${binary} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(${binary} PRIVATE iphlpapi ws2_32)
endforeach()
target_link_libraries(diagnostics_core PRIVATE psapi ole32 winsqlite3 winhttp setupapi advapi32 wlanapi bcrypt crypt32 uuid nlohmann_json::nlohmann_json ZLIB::ZLIBSTATIC)
option(VIPNET_ACCEPTANCE "Build the loopback-only transport fixture" OFF)
if(VIPNET_ACCEPTANCE)
  add_executable(vipnet-chart-check speed_chart_check.cpp speed_chart.cpp)
  target_compile_features(vipnet-chart-check PRIVATE cxx_std_20)
  target_compile_definitions(vipnet-chart-check PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(vipnet-chart-check PRIVATE /W4 /utf-8)
  add_executable(vipnet-loopback-fixture entry.cpp sender.cpp)
  target_compile_features(vipnet-loopback-fixture PRIVATE cxx_std_20)
  target_compile_definitions(vipnet-loopback-fixture PRIVATE VIPNET_LOOPBACK_RELAY WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(vipnet-loopback-fixture PRIVATE diagnostics_core ZLIB::ZLIBSTATIC)
endif()
